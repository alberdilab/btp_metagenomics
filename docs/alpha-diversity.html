Chapter 7 Alpha diversity

We quantify within-sample diversity using Hill numbers: richness (q0), neutral/abundance-weighted (q1), phylogenetic (q1 + tree), and functional (q1 + traits), and compare these across environments and sex.

7.1.1 Hill numbers

Richness of microbiotic species in samples

Note: Calculate four types of Hill number diversity metrics for each sample: richness (q=0, species count unweighted by abundance), neutral (q=1, Shannon diversity abundance-weighted), phylogenetic (q=1 + tree, incorporates phylogenetic relationships), and functional (q=1 + traits, incorporates functional trait distances). Then merge all metrics into a single alpha_div dataframe.

# Calculate Hill numbers
richness <- genome_counts_filt %>%
  column_to_rownames(var = "genome") %>%
  dplyr::select(where(~ !all(. == 0))) %>%
  hilldiv(., q = 0) %>%
  t() %>%
  as.data.frame() %>%
  dplyr::rename(richness = 1) %>%
  rownames_to_column(var = "sample")

neutral <- genome_counts_filt %>%
  column_to_rownames(var = "genome") %>%
  dplyr::select(where(~ !all(. == 0))) %>%
  hilldiv(., q = 1) %>%
  t() %>%
  as.data.frame() %>%
  dplyr::rename(neutral = 1) %>%
  rownames_to_column(var = "sample")

phylogenetic <- genome_counts_filt %>%
  column_to_rownames(var = "genome") %>%
  dplyr::select(where(~ !all(. == 0))) %>%
  hilldiv(., q = 1, tree = genome_tree_pruned) %>%
  t() %>%
  as.data.frame() %>%
  dplyr::rename(phylogenetic = 1) %>%
  rownames_to_column(var = "sample")

# Aggregate basal GIFT into elements
dist <- genome_gifts %>%
  to.elements(., GIFT_db) %>%
  traits2dist(., method = "gower")

functional <- genome_counts_filt %>%
  filter(genome %in% rownames(dist)) %>% 
  column_to_rownames(var = "genome") %>%
  dplyr::select(where(~ !all(. == 0))) %>%
  hilldiv(., q = 1, dist = dist) %>%
  t() %>%
  as.data.frame() %>%
  dplyr::rename(functional = 1) %>%
  rownames_to_column(var = "sample") %>%
  mutate(functional = if_else(is.nan(functional), 1, functional))

# Merge all metrics
alpha_div <- richness %>%
  full_join(neutral, by = join_by(sample == sample)) %>%
  full_join(phylogenetic, by = join_by(sample == sample)) %>%
  full_join(functional, by = join_by(sample == sample))

make_plot <- function(target_metric, ylab){
  alpha_div %>%
    pivot_longer(-sample, names_to = "metric", values_to = "value") %>%
    inner_join(sample_metadata, by = dplyr::join_by(sample == sample)) %>%
    filter(metric == target_metric) %>%
    ggplot(aes(x = broad_environment, y = value)) +
    geom_boxplot(aes(fill = broad_environment),
                 width = 0.65, alpha = 0.75, outlier.shape = NA,
                 show.legend = TRUE, linewidth = 0.6) +
    geom_jitter(aes(color = broad_environment),
                width = 0.3, size = 1.8, alpha = 0.65,
                show.legend = FALSE) +
    scale_fill_manual(values = treatment_colors, labels = env_labels_long,
                      breaks = env_limits, drop = FALSE) +
    scale_color_manual(values = treatment_colors, guide = "none") +
    scale_x_discrete(limits = env_limits, labels = env_labels_short, drop = FALSE) +
    # p-value: white label at the very top of the panel
    stat_compare_means(method = "wilcox.test", label = "p.format",
                       geom = "label", fill = "white", label.size = 0.4,
                       label.padding = unit(0.2, "lines"),
                       label.r = unit(0.15, "lines"),
                       color = "black",
                       label.y.npc = 0.98, label.x = 3, size = 4, fontface = "bold") +
    scale_y_continuous(expand = expansion(mult = c(0.08, 0.12))) +
    coord_cartesian(clip = "off") +
    theme_classic() +
    theme(
      legend.title = element_blank(),
      legend.text = element_text(size = 11),
      legend.position = "top",
      legend.margin = margin(b = 6, t = 0),
      legend.key.size = unit(0.8, "lines"),
      legend.spacing.x = unit(0.3, "cm"),
      legend.box.margin = margin(b = 5),

      plot.margin = margin(t = 8, r = 12, b = 10, l = 12),
      panel.spacing = unit(1.2, "lines")
    ) +
    labs(y = ylab)
}

Defines color scheme for sex categories, creates a plotting function that facets by environment, generates four plots (richness, neutral, phylogenetic, functional), combines plots into a 2x2 grid with shared legend, and includes Wilcoxon test p-values comparing Female vs Male within each environment.

# colors (keep or replace)
sex_colors <- c("Female" = "#E41A1C", "Male" = "#377EB8", "Unknown" = "grey50")

make_plot_sex <- function(target_metric, ylab) {
  alpha_div %>%
    pivot_longer(-sample, names_to = "metric", values_to = "value") %>%
    inner_join(sample_metadata, by = dplyr::join_by(sample == sample)) %>%
    filter(metric == target_metric) %>%
    mutate(
      # ensure order + map short ticks
      sex = factor(sex, levels = c("Female","Male","Unknown"))
    ) %>%
    ggplot(aes(x = sex, y = value, fill = sex)) +
    geom_boxplot(width = 0.6, alpha = 0.7, outlier.shape = NA, 
                 show.legend = TRUE, linewidth = 0.5) +
    geom_jitter(aes(color = sex), width = 0.2, size = 1.3, alpha = 0.7, 
                show.legend = FALSE) +
    facet_grid(
      ~ broad_environment,
      labeller = labeller(broad_environment = c(
        "1000198 - Mixed forest"       = "MF",
        "1000221 - Temperate woodland" = "TW",
        "1000218 - Xeric shrubland"    = "XS",
        "1000215 - Temperate shrubland" = "TS",
        "1000245 - Cropland"           = "C"
      )),
      scales = "fixed", space = "free_x"
    ) +
    # p-values: readable white labels at the top of each facet
    ggpubr::stat_compare_means(
      method = "wilcox.test",
      comparisons = list(c("Female","Male")),
      label = "p.format",
      geom = "label", fill = "white", color = "black",
      label.size = 0.3, label.padding = grid::unit(0.15, "lines"),
      label.r = grid::unit(0.1, "lines"),
      label.y.npc = 0.98, label.x.npc = 0.5, size = 3.5, fontface = "bold"
    ) +
    # SHORT x-ticks + legend with full names
    scale_x_discrete(labels = c(Female = "F", Male = "M", Unknown = "U")) +
    scale_fill_manual(
      values = sex_colors,
      labels = c("Female (F)", "Male (M)", "Unknown (U)")
    ) +
    scale_color_manual(values = sex_colors, guide = "none") +
    scale_y_continuous(expand = expansion(mult = c(0.05, 0.15))) +
    coord_cartesian(clip = "off") +
    theme_classic() +
    theme(
      legend.position = "top",
      legend.title = element_blank(),

      legend.margin = margin(b = 8),

    ) +
    labs(
      y = ylab,
      caption = "Axis codes: F=Female; M=Male; U=Unknown."
